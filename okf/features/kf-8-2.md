# KF 8.2 — Konfigurasi i18n Bilingual

Nuxt config setting up bilingual route mappings using `@nuxtjs/i18n` and `@nuxt/content` source drivers.

## `docs/nuxt.config.ts`

```ts
export default defineNuxtConfig({
  extends: ['@nuxt/ui-pro'],

  modules: [
    '@nuxt/content',
    '@nuxtjs/i18n',
  ],

  i18n: {
    locales: [
      {
        code: 'en',
        language: 'en-US',
        name: 'English',
        flag: '🇺🇸',
        files: ['en.json']
      },
      {
        code: 'id',
        language: 'id-ID',
        name: 'Indonesia',
        flag: '🇮🇩',
        files: ['id.json']
      },
    ],
    defaultLocale: 'en',
    strategy: 'prefix_except_default',
    // English (default) -> /getting-started
    // Indonesian (prefix) -> /id/memulai
  },

  content: {
    sources: {
      content: {
        driver: 'fs',
        prefix: '/docs',
        base: './content'
      }
    }
  },

  ui: {
    icons: ['heroicons', 'simple-icons']
  }
})
```
